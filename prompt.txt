import express from 'express';
import { createServer } from 'http';
import { Server } from 'socket.io';
const app = express();
const server = createServer(app);
const io = new Server(server, { cors: { origin: 'http://localhost:3002' } }); // Adjust for your frontend port

let rooms = [];
let users = [{ id: '1', username: 'Sensei', email: 'sensei@example.com', password: 'pass123' }];

io.on('connection', (socket) => {
  console.log('User connected:', socket.id);

  socket.on('register', ({ username, email, password }, callback) => {
    if (users.some((u) => u.email === email)) return callback({ success: false, error: 'Email exists' });
    const user = { id: socket.id, username, email, password };
    users.push(user);
    callback({ success: true, user });
  });

  socket.on('login', ({ email, password }, callback) => {
    const user = users.find((u) => u.email === email && u.password === password);
    if (!user) return callback({ success: false, error: 'Invalid credentials' });
    // Update the user's id to the current socket id.
    callback({ success: true, user: { ...user, id: socket.id } });
  });

  socket.on('createRoom', ({ title, isPublic }, callback) => {
    console.log('Received createRoom:', { title, isPublic });
    // Generate a unique room ID instead of relying solely on Date.now()
    const roomId = `${Date.now()}-${Math.random().toString(36).substr(2, 5)}`;
    const room = {
      id: roomId,
      title,
      hostId: socket.id,
      participants: [{ userId: socket.id, isSpeaker: true, isMuted: false }],
      isPublic,
    };
    rooms.push(room);
    socket.join(roomId);
    io.emit('roomsUpdated', rooms);
    console.log('Sending createRoom callback:', roomId);
    callback(roomId);
  });

  socket.on('joinRoom', (roomId, callback) => {
    const room = rooms.find((r) => r.id === roomId);
    if (!room || room.participants.some((p) => p.userId === socket.id)) {
      io.to(roomId).emit('roomUpdated', room);
      io.emit('roomsUpdated', rooms);
      callback(true);
      return;
    }
    room.participants.push({ userId: socket.id, isSpeaker: false, isMuted: false });
    socket.join(roomId);
    io.to(roomId).emit('roomUpdated', room);
    io.emit('roomsUpdated', rooms);
    callback(true);
  });

  socket.on('leaveRoom', (roomId) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room) {
      room.participants = room.participants.filter((p) => p.userId !== socket.id);
      socket.leave(roomId);
      io.to(roomId).emit('roomUpdated', room);
      io.emit('roomsUpdated', rooms);
    }
  });

  socket.on('toggleMute', ({ roomId, userId }) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room) {
      room.participants = room.participants.map((p) =>
        p.userId === userId ? { ...p, isMuted: !p.isMuted } : p
      );
      io.to(roomId).emit('roomUpdated', room);
    }
  });

  socket.on('promoteToSpeaker', ({ roomId, userId }) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room && room.hostId === socket.id) {
      room.participants = room.participants.map((p) =>
        p.userId === userId ? { ...p, isSpeaker: true, isMuted: false } : p
      );
      io.to(roomId).emit('roomUpdated', room);
    }
  });

  socket.on('kickUser', ({ roomId, userId }) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room && room.hostId === socket.id) {
      room.participants = room.participants.filter((p) => p.userId !== userId);
      io.to(roomId).emit('roomUpdated', room);
      io.emit('roomsUpdated', rooms);
    }
  });

  socket.on('offer', ({ roomId, offer }) => {
    socket.to(roomId).emit('offer', { offer, from: socket.id });
  });

  socket.on('answer', ({ roomId, answer, to }) => {
    socket.to(to).emit('answer', { answer, from: socket.id });
  });

  socket.on('ice-candidate', ({ roomId, candidate }) => {
    socket.to(roomId).emit('ice-candidate', { candidate, from: socket.id });
  });

  socket.on('disconnect', () => {
    rooms = rooms.map((r) => ({
      ...r,
      participants: r.participants.filter((p) => p.userId !== socket.id),
    }));
    io.emit('roomsUpdated', rooms);
    console.log('User disconnected:', socket.id);
  });
});

server.listen(3001, () => console.log('Server running on port 3001'));

import React, { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import { io, Socket } from 'socket.io-client';

interface User {
  id: string;
  username: string;
  email: string;
}

interface Room {
  id: string;
  title: string;
  hostId: string;
  participants: { userId: string; isSpeaker: boolean; isMuted: boolean }[];
  isPublic: boolean;
}

interface AppContextType {
  currentUser: User | null;
  rooms: Room[];
  login: (email: string, password: string, callback: (result: { success: boolean; error?: string }) => void) => void;
  register: (username: string, email: string, password: string, callback: (result: { success: boolean; error?: string }) => void) => void;
  logout: () => void;
  createRoom: (title: string, isPublic: boolean, callback: (roomId: string | null) => void) => void;
  joinRoom: (roomId: string, callback: (success: boolean) => void) => void;
  leaveRoom: (roomId: string) => void;
  toggleMute: (roomId: string, userId: string) => void;
  promoteToSpeaker: (roomId: string, userId: string) => void;
  kickUser: (roomId: string, userId: string) => void;
  socket: Socket | null;
  isConnected: boolean;
}

const AppContext = createContext<AppContextType | undefined>(undefined);

export const AppProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
  const [currentUser, setCurrentUser] = useState<User | null>(null);
  const [rooms, setRooms] = useState<Room[]>([]);
  const [isConnected, setIsConnected] = useState(false);
  const [socket] = useState<Socket>(() =>
    io('http://localhost:3001', {
      autoConnect: false,
      reconnection: true,
      reconnectionAttempts: 5,
      reconnectionDelay: 1000,
    })
  );

  useEffect(() => {
    console.log('Initializing Socket.IO client...');
    socket.connect();

    socket.on('connect', () => {
      console.log('Socket connected successfully:', socket.id);
      setIsConnected(true);
    });

    socket.on('disconnect', (reason) => {
      console.log('Socket disconnected:', reason);
      setIsConnected(false);
    });

    socket.on('connect_error', (err) => {
      console.error('Socket connection error:', err.message);
      setIsConnected(false);
    });

    socket.on('reconnect', (attempt) => {
      console.log('Socket reconnected after', attempt, 'attempts');
      setIsConnected(true);
    });

    socket.on('roomsUpdated', (updatedRooms: Room[]) => {
      console.log('Received roomsUpdated:', updatedRooms);
      setRooms(updatedRooms);
    });

    return () => {
      console.log('Cleaning up Socket.IO listeners...');
      socket.off('connect');
      socket.off('disconnect');
      socket.off('connect_error');
      socket.off('reconnect');
      socket.off('roomsUpdated');
      socket.disconnect();
    };
  }, [socket]);

  const login = (email: string, password: string, callback: (result: { success: boolean; error?: string }) => void) => {
    if (!isConnected) return callback({ success: false, error: 'Not connected to server' });
    socket.emit('login', { email, password }, ({ success, user, error }: any) => {
      console.log('Login response:', { success, user, error });
      if (success) {
        setCurrentUser(user);
        callback({ success: true });
      } else {
        callback({ success: false, error });
      }
    });
  };

  const register = (
    username: string,
    email: string,
    password: string,
    callback: (result: { success: boolean; error?: string }) => void
  ) => {
    if (!isConnected) return callback({ success: false, error: 'Not connected to server' });
    socket.emit('register', { username, email, password }, ({ success, user, error }: any) => {
      console.log('Register response:', { success, user, error });
      if (success) {
        setCurrentUser(user);
        callback({ success: true });
      } else {
        callback({ success: false, error });
      }
    });
  };

  const logout = () => {
    setCurrentUser(null);
    socket.disconnect();
  };

  const createRoom = (title: string, isPublic: boolean, callback: (roomId: string | null) => void) => {
    if (!currentUser || !isConnected) {
      console.log('createRoom failed: User not logged in or socket not connected');
      return callback(null);
    }
    console.log('Emitting createRoom:', { title, isPublic });
    socket.emit('createRoom', { title, isPublic }, (roomId: string) => {
      console.log('createRoom callback received:', roomId);
      callback(roomId);
    });
  };

  const joinRoom = (roomId: string, callback: (success: boolean) => void) => {
    if (!currentUser || !isConnected) return callback(false);
    socket.emit('joinRoom', roomId, (success: boolean) => {
      console.log('Join room response:', success);
      callback(success);
    });
  };

  const leaveRoom = (roomId: string) => {
    if (!isConnected) return;
    socket.emit('leaveRoom', roomId);
  };

  const toggleMute = (roomId: string, userId: string) => {
    if (!isConnected) return;
    socket.emit('toggleMute', { roomId, userId });
  };

  const promoteToSpeaker = (roomId: string, userId: string) => {
    if (!isConnected) return;
    socket.emit('promoteToSpeaker', { roomId, userId });
  };

  const kickUser = (roomId: string, userId: string) => {
    if (!isConnected) return;
    socket.emit('kickUser', { roomId, userId });
  };

  return (
    <AppContext.Provider
      value={{
        currentUser,
        rooms,
        login,
        register,
        logout,
        createRoom,
        joinRoom,
        leaveRoom,
        toggleMute,
        promoteToSpeaker,
        kickUser,
        socket,
        isConnected,
      }}
    >
      {children}
    </AppContext.Provider>
  );
};

export const useAppContext = () => {
  const context = useContext(AppContext);
  if (!context) throw new Error('useAppContext must be used within AppProvider');
  return context;
};

import React, { useEffect, useRef, useState } from 'react';
import UserAvatar from '../components/UserAvatar';
import ControlButton from '../components/ControlButton';
import { Mic, MicOff, PhoneOff, UserMinus, UserPlus } from 'lucide-react';
import { useAppContext } from '../context/AppContext';
import { useParams, useNavigate } from 'react-router-dom';

const LiveRoomPage: React.FC = () => {
  const { roomId } = useParams<{ roomId: string }>();
  const { currentUser, rooms, socket, toggleMute, promoteToSpeaker, kickUser, leaveRoom, isConnected } = useAppContext();
  const navigate = useNavigate();
  const [room, setRoom] = useState<any>(null);
  const localStream = useRef<MediaStream | null>(null);
  const peers = useRef<{ [key: string]: RTCPeerConnection }>({});

  useEffect(() => {
    if (!socket || !roomId || !isConnected || !currentUser) {
      console.log('Cannot initialize room: Missing socket, roomId, connection, or user');
      return;
    }
    
    console.log('Initializing LiveRoomPage for roomId:', roomId);

    // Set initial room from rooms array if available
    const initialRoom = rooms.find((r) => r.id === roomId);
    if (initialRoom) {
      console.log('Found room in initial rooms:', initialRoom);
      setRoom(initialRoom);
    }

    // Join room if not already joined
    const joinIfNeeded = () => {
      const currentRoom = rooms.find((r) => r.id === roomId);
      if (!currentRoom || !currentRoom.participants.some((p: any) => p.userId === currentUser.id)) {
        console.log('Joining room:', roomId);
        socket.emit('joinRoom', roomId, (success: boolean) => {
          if (success) {
            console.log('Successfully joined room:', roomId);
          } else {
            console.log('Failed to join room:', roomId);
            navigate('/'); // Redirect if join fails
          }
        });
      } else {
        console.log('Already in room:', roomId);
      }
    };

    joinIfNeeded();

    // Create individual callback functions for events
    const handleRoomUpdated = (updatedRoom: any) => {
      console.log('Received roomUpdated:', updatedRoom);
      if (updatedRoom.id === roomId) {
        setRoom(updatedRoom);
      }
    };

    const handleRoomsUpdated = (updatedRooms: any[]) => {
      console.log('Received roomsUpdated:', updatedRooms);
      const updatedRoom = updatedRooms.find((r) => r.id === roomId);
      if (updatedRoom) {
        setRoom(updatedRoom);
      }
    };

    const handleOffer = async ({ offer, from }: any) => {
      const pc = createPeerConnection(from);
      await pc.setRemoteDescription(new RTCSessionDescription(offer));
      const answer = await pc.createAnswer();
      await pc.setLocalDescription(answer);
      socket.emit('answer', { roomId, answer, to: from });
    };

    const handleAnswer = async ({ answer, from }: any) => {
      const pc = peers.current[from];
      if (pc) await pc.setRemoteDescription(new RTCSessionDescription(answer));
    };

    const handleIceCandidate = async ({ candidate, from }: any) => {
      const pc = peers.current[from];
      if (pc) await pc.addIceCandidate(new RTCIceCandidate(candidate));
    };

    // Register event listeners specific to this component
    socket.on('roomUpdated', handleRoomUpdated);
    socket.on('roomsUpdated', handleRoomsUpdated);
    socket.on('offer', handleOffer);
    socket.on('answer', handleAnswer);
    socket.on('ice-candidate', handleIceCandidate);

    const initWebRTC = async () => {
      try {
        localStream.current = await navigator.mediaDevices.getUserMedia({ audio: true });
        const roomData = rooms.find((r) => r.id === roomId);
        if (!roomData) return;

        roomData.participants.forEach((p: any) => {
          if (p.userId !== currentUser.id && p.isSpeaker) createPeerConnection(p.userId);
        });
      } catch (err) {
        console.error('WebRTC init error:', err);
      }
    };

    initWebRTC();

    function createPeerConnection(userId: string) {
      const pc = new RTCPeerConnection({
        iceServers: [{ urls: 'stun:stun.l.google.com:19302' }],
      });
      peers.current[userId] = pc;

      if (localStream.current) {
        localStream.current.getTracks().forEach((track) => pc.addTrack(track, localStream.current!));
      }

      pc.ontrack = (event) => {
        const audio = document.createElement('audio');
        audio.srcObject = event.streams[0];
        audio.autoplay = true;
        document.body.appendChild(audio);
      };

      pc.onicecandidate = (event) => {
        if (event.candidate) {
          socket!.emit('ice-candidate', { roomId, candidate: event.candidate });
        }
      };

      pc.createOffer()
        .then((offer) => pc.setLocalDescription(offer))
        .then(() => socket!.emit('offer', { roomId, offer: pc.localDescription }));

      return pc;
    }

    return () => {
      console.log('Cleaning up LiveRoomPage...');
      // Remove only the listeners that this component added.
      socket.off('roomUpdated', handleRoomUpdated);
      socket.off('roomsUpdated', handleRoomsUpdated);
      socket.off('offer', handleOffer);
      socket.off('answer', handleAnswer);
      socket.off('ice-candidate', handleIceCandidate);

      if (localStream.current) {
        localStream.current.getTracks().forEach((track) => track.stop());
      }
      Object.values(peers.current).forEach((pc) => pc.close());
    };
  }, [socket, roomId, currentUser, rooms, isConnected, navigate]);

  if (!currentUser || !isConnected) {
    return (
      <div className="min-h-screen bg-gray-900 text-white flex items-center justify-center">
        {isConnected ? 'Loading room...' : 'Not connected to server. Please refresh.'}
      </div>
    );
  }

  if (!room) {
    return (
      <div className="min-h-screen bg-gray-900 text-white flex items-center justify-center">
        Room not found or still loading...
      </div>
    );
  }

  const isHost = room.hostId === currentUser.id;
  const currentParticipant = room.participants.find((p: any) => p.userId === currentUser.id);

  const handleLeave = () => {
    leaveRoom(room.id);
    navigate('/');
  };

  const handleToggleMute = () => {
    toggleMute(room.id, currentUser.id);
    if (localStream.current) {
      localStream.current.getAudioTracks()[0].enabled = !currentParticipant?.isMuted;
    }
  };

  const handleKick = (userId: string) => {
    if (isHost) kickUser(room.id, userId);
  };

  const handlePromote = (userId: string) => {
    if (isHost) promoteToSpeaker(room.id, userId);
  };

  return (
    <div className="min-h-screen flex flex-col bg-gray-900 bg-screentone">
      <header
        className="bg-manga-black p-3 text-manga-white relative overflow-hidden"
        style={{ clipPath: 'polygon(0 0, 100% 0, 100% 85%, 0% 100%)' }}
      >
        <div className="absolute bottom-0 left-0 h-1 w-full bg-neon-magenta"></div>
        <h1 className="font-manga font-black text-2xl">{room.title}</h1>
        <p className="text-sm text-gray-400">
          Hosted by {room.hostId === currentUser.id ? 'You' : `User_${room.hostId.slice(0, 4)}`}
        </p>
      </header>
      <main className="flex-grow p-4 md:p-8 grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4 md:gap-6 place-items-center overflow-y-auto">
        {room.participants.map((p: any) => (
          <div key={p.userId} className="relative">
            <UserAvatar
              name={p.userId === currentUser.id ? currentUser.username : `User_${p.userId.slice(0, 4)}`}
              status={
                p.isSpeaker && !p.isMuted ? 'speaking' : p.isMuted ? 'muted' : 'listening'
              }
              isHost={p.userId === room.hostId}
            />
            {isHost && p.userId !== currentUser.id && !p.isSpeaker && (
              <ControlButton
                icon={<UserPlus size={16} />}
                label="Promote"
                variant="cyan"
                onClick={() => handlePromote(p.userId)}
              />
            )}
          </div>
        ))}
      </main>
      <footer className="bg-manga-black p-3 border-t-4 border-manga-black flex justify-center items-center flex-wrap">
        <ControlButton
          icon={currentParticipant?.isMuted ? <MicOff size={24} /> : <Mic size={24} />}
          label={currentParticipant?.isMuted ? 'Unmute' : 'Mute'}
          onClick={handleToggleMute}
          variant={currentParticipant?.isMuted ? 'magenta' : 'lime'}
          isActive={currentParticipant?.isMuted}
        />
        {isHost && (
          <ControlButton
            icon={<UserMinus size={24} />}
            label="Kick"
            variant="magenta"
            onClick={() =>
              handleKick(
                room.participants.find((p: any) => p.userId !== currentUser.id)?.userId || ''
              )
            }
          />
        )}
        <ControlButton
          icon={<PhoneOff size={24} />}
          label="Leave"
          variant="cyan"
          onClick={handleLeave}
        />
      </footer>
    </div>
  );
};

// File: /server/index.js
import express from 'express';
import { createServer } from 'http';
import { Server } from 'socket.io';
const app = express();
const server = createServer(app);
const io = new Server(server, { cors: { origin: 'http://localhost:3002' } }); // Adjust for your frontend port

let rooms = [];
let users = [{ id: '1', username: 'Sensei', email: 'sensei@example.com', password: 'pass123' }];

io.on('connection', (socket) => {
  console.log('User connected:', socket.id);

  socket.on('register', ({ username, email, password }, callback) => {
    if (users.some((u) => u.email === email)) return callback({ success: false, error: 'Email exists' });
    const user = { id: socket.id, username, email, password };
    users.push(user);
    callback({ success: true, user });
  });

  socket.on('login', ({ email, password }, callback) => {
    const user = users.find((u) => u.email === email && u.password === password);
    if (!user) return callback({ success: false, error: 'Invalid credentials' });
    // Update the user's id to the current socket id.
    callback({ success: true, user: { ...user, id: socket.id } });
  });

  socket.on('createRoom', ({ title, isPublic }, callback) => {
    console.log('Received createRoom:', { title, isPublic });
    // Generate a unique room ID instead of relying solely on Date.now()
    const roomId = `${Date.now()}-${Math.random().toString(36).substr(2, 5)}`;
    const room = {
      id: roomId,
      title,
      hostId: socket.id,
      participants: [{ userId: socket.id, isSpeaker: true, isMuted: false }],
      isPublic,
    };
    rooms.push(room);
    socket.join(roomId);
    io.emit('roomsUpdated', rooms);
    console.log('Sending createRoom callback:', roomId);
    callback(roomId);
  });

  socket.on('joinRoom', (roomId, callback) => {
    const room = rooms.find((r) => r.id === roomId);
    if (!room || room.participants.some((p) => p.userId === socket.id)) {
      io.to(roomId).emit('roomUpdated', room);
      io.emit('roomsUpdated', rooms);
      callback(true);
      return;
    }
    room.participants.push({ userId: socket.id, isSpeaker: false, isMuted: false });
    socket.join(roomId);
    io.to(roomId).emit('roomUpdated', room);
    io.emit('roomsUpdated', rooms);
    callback(true);
  });

  socket.on('leaveRoom', (roomId) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room) {
      room.participants = room.participants.filter((p) => p.userId !== socket.id);
      socket.leave(roomId);
      io.to(roomId).emit('roomUpdated', room);
      io.emit('roomsUpdated', rooms);
    }
  });

  socket.on('toggleMute', ({ roomId, userId }) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room) {
      room.participants = room.participants.map((p) =>
        p.userId === userId ? { ...p, isMuted: !p.isMuted } : p
      );
      io.to(roomId).emit('roomUpdated', room);
    }
  });

  socket.on('promoteToSpeaker', ({ roomId, userId }) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room && room.hostId === socket.id) {
      room.participants = room.participants.map((p) =>
        p.userId === userId ? { ...p, isSpeaker: true, isMuted: false } : p
      );
      io.to(roomId).emit('roomUpdated', room);
    }
  });

  socket.on('kickUser', ({ roomId, userId }) => {
    const room = rooms.find((r) => r.id === roomId);
    if (room && room.hostId === socket.id) {
      room.participants = room.participants.filter((p) => p.userId !== userId);
      io.to(roomId).emit('roomUpdated', room);
      io.emit('roomsUpdated', rooms);
    }
  });

  socket.on('offer', ({ roomId, offer }) => {
    socket.to(roomId).emit('offer', { offer, from: socket.id });
  });

  socket.on('answer', ({ roomId, answer, to }) => {
    socket.to(to).emit('answer', { answer, from: socket.id });
  });

  socket.on('ice-candidate', ({ roomId, candidate }) => {
    socket.to(roomId).emit('ice-candidate', { candidate, from: socket.id });
  });

  socket.on('disconnect', () => {
    rooms = rooms.map((r) => ({
      ...r,
      participants: r.participants.filter((p) => p.userId !== socket.id),
    }));
    io.emit('roomsUpdated', rooms);
    console.log('User disconnected:', socket.id);
  });
});

server.listen(3001, () => console.log('Server running on port 3001'));
